<script>
  const form = document.getElementById('calcForm');
  const result = document.getElementById('result');
  const tabs = [...document.querySelectorAll('.seg-btn')];
  const fields = {
    distance: document.querySelector('[data-field="distance"]'),
    speed:    document.querySelector('[data-field="speed"]'),
    time:     document.querySelector('[data-field="time"]')
  };

  // Ποιο πεδίο μένει κενό (αυτό που υπολογίζουμε) ανά mode
  const solveFor = { speed: 'speed', distance: 'distance', time: 'time' };
  let mode = 'speed';

  function render() {
    tabs.forEach(t => t.setAttribute('aria-selected', String(t.dataset.mode === mode)));
    Object.entries(fields).forEach(([key, el]) => {
      el.classList.toggle('is-hidden', key === solveFor[mode]);
    });
  }

  tabs.forEach(tab => tab.addEventListener('click', () => {
    mode = tab.dataset.mode;
    render();
  }));

  form.addEventListener('submit', e => {
    e.preventDefault();
    const num = id => parseFloat(document.getElementById(id).value);
    const d = num('distance'), s = num('speed'), t = num('time');
    const target = solveFor[mode];
    const needed = target === 'speed' ? [d, t] : target === 'distance' ? [s, t] : [d, s];

    if (needed.some(v => !isFinite(v) || v <= 0)) {
      return show('Συμπλήρωσε σωστά όλα τα πεδία.', true);
    }

    let value, unit;
    if (target === 'speed')     { value = d / t;             unit = 'χλμ/ώ'; }
    else if (target === 'distance') { value = s * t;         unit = 'χλμ'; }
    else                        { value = d / s;             unit = 'ώρες'; }

    show(`Αποτέλεσμα: <strong>${value.toFixed(2)}</strong> ${unit}`, false);
  });

  function show(html, isError) {
    result.style.display = 'flex';
    result.classList.toggle('is-error', isError);
    result.innerHTML = isError ? html : html;
  }

  render();
</script>
</body>
</html>

**Τα χρώματα που πρόσθεσα (και πού πάνε):**

| Ρόλος | Variable | Τιμή |
|---|---|---|
| Φόντο σελίδας / card | `--bg-page`, `--bg-card` | `#eef2f7`, `#fff` |
| Κύριο (buttons, active tab) | `--primary` | `#4f46e5` indigo |
| Έμφαση μονάδες | `--accent` | `#06b6d4` cyan |
| Κείμενα | `--text-strong/body/muted` | slate scale |
| Borders / focus ring | `--border`, `--ring` | `#dbe1ea`, indigo 18% |
| Επιτυχία / σφάλμα | `--success`, `--error` | `#16a34a`, `#dc2626` |

// SpeedCalculator.jsx — React version (ίδια palette, dark mode toggle)
import { useState } from 'react';

const MODES = [
  { id: 'speed',    label: 'Απόσταση' },
  { id: 'distance', label: 'Ταχύτητα' },
  { id: 'time',     label: 'Χρόνος' },
];

export default function SpeedCalculator() {
  const [mode, setMode] = useState('speed');
  const [values, setValues] = useState({ distance: '', speed: '', time: '' });
  const [result, setResult] = useState(null); // { text, unit } | { error }
  const [dark, setDark] = useState(false);

  const solveFor = { speed: 'speed', distance: 'distance', time: 'time' }[mode];
  const fields = [
    { key: 'distance', label: 'Απόσταση', unit: 'χλμ' },
    { key: 'speed',    label: 'Ταχύτητα', unit: 'χλμ/ώ' },
    { key: 'time',     label: 'Χρόνος',   unit: 'ώρες' },
  ];

  const submit = (e) => {
    e.preventDefault();
    const n = (k) => parseFloat(values[k]);
    const needed = solveFor === 'speed' ? ['distance','time']
                 : solveFor === 'distance' ? ['speed','time']
                 : ['distance','speed'];

    if (needed.some(k => !isFinite(n(k)) || n(k) <= 0)) {
      return setResult({ error: 'Συμπλήρωσε σωστά όλα τα πεδία.' });
    }

    if (solveFor === 'speed') {
      setResult({ text: (n('distance') / n('time')).toFixed(2), unit: 'χλμ/ώ' });
    } else if (solveFor === 'distance') {
      setResult({ text: (n('speed') * n('time')).toFixed(2), unit: 'χλμ' });
    } else {
      setResult({ text: (n('distance') / n('speed')).toFixed(2), unit: 'ώρες' });
    }
  };

  return (
    <div className={`calc-card ${dark ? 'is-dark' : ''}`}>
      <button type="button" className="theme-toggle" onClick={() => setDark(d => !d)}>
        {dark ? '☀️ Φωτεινό' : '🌙 Σκοτεινό'}
      </button>

      <h2 className="calc-title">Επέλεξε τον τρόπο υπολογισμού</h2>
      <p className="calc-subtitle">
        Υπολογισμός ταχύτητας για μια συγκεκριμένη απόσταση σε χιλιόμετρα, με βάση τη
        χιλιομετρική απόσταση και τον χρόνο που χρειάζεται να τη διανύσεις.
      </p>

      <div className="seg" role="tablist">
        {MODES.map(m => (
          <button key={m.id} type="button" role="tab" className="seg-btn"
            aria-selected={mode === m.id} onClick={() => setMode(m.id)}>
            {m.label}
          </button>
        ))}
      </div>

      <form onSubmit={submit} noValidate>
        {fields.filter(f => f.key !== solveFor).map(f => (
          <div className="field" key={f.key}>
            <label htmlFor={f.key}>{f.label}</label>
            <div className="input-wrap">
              <input id={f.key} type="number" min="0" step="any"
                value={values[f.key]}
                onChange={e => setValues(v => ({ ...v, [f.key]: e.target.value }))} />
              <span className="unit">{f.unit}</span>
            </div>
          </div>
        ))}

        <button type="submit" className="btn-calc">Υπολογισμός</button>

        {result && (
          <output className={`result ${result.error ? 'is-error' : ''}`}>
            {result.error ? result.error
              : <>Αποτέλεσμα: <strong>{result.text}</strong> {result.unit}</>}
          </output>
        )}
      </form>
    </div>
  );
}

**Σημείωση για τα tabs:** στο UI η σειρά είναι "Απόσταση / Ταχύτητα / Χρόνος", αλλά το κάθε tab επιλέγει *τι λύνεις*. Αν θες η ετικέτα να δείχνει πάντα το αγνώστοτο (π.χ. tab "Ταχύτητα" → βρίσκει ταχύτητα), άσε το `solveFor` map ως έχει — απλά άλλαξε τα `label` στο `MODES`.
